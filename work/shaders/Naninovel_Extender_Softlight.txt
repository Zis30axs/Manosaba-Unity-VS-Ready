{

    u_xlat1 = texture(_MainTex, vs_TEXCOORD0.xy);
    u_xlat0.x = u_xlat0.x + (-u_xlat1.w);
    u_xlat0.x = _EnableExternalAlpha * u_xlat0.x + u_xlat1.w;

    u_xlat4.xyz = u_xlat4.xyz * u_xlat2.xyz;
    u_xlat2.xyz = (-vs_COLOR0.xyz) + vec3(1.0, 1.0, 1.0);
    u_xlat3.xyz = u_xlat1.xyz + u_xlat1.xyz;
    u_xlat1.xyz = u_xlat1.xyz * u_xlat1.xyz;
    u_xlat4.xyz = u_xlat3.xyz * u_xlat2.xyz + u_xlat4.xyz;
    u_xlat2.xyz = (-vs_COLOR0.xyz) * vec3(2.0, 2.0, 2.0) + vec3(1.0, 1.0, 1.0);
    u_xlat1.xyz = u_xlat1.xyz * u_xlat2.xyz;
    u_xlat1.xyz = u_xlat3.xyz * vs_COLOR0.xyz + u_xlat1.xyz;
    u_xlatb2.xyz = lessThan(vs_COLOR0.xyzx, vec4(0.5, 0.5, 0.5, 0.0)).xyz;

        vec3 hlslcc_movcTemp = u_xlat4;

        hlslcc_movcTemp.z = (u_xlatb2.z) ? u_xlat1.z : u_xlat4.z;
        u_xlat4 = hlslcc_movcTemp;
    }
    SV_Target0.xyz = u_xlat0.xxx * u_xlat4.xyz;
    SV_Target0.w = u_xlat0.x;
    return;
}
